jobs:
  accessibility:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 24

      - run: npm ci
      - run: npx playwright install chromium

      - name: Start the app
        run: npm run start &

      - run: npx a1 doctor --strict
      - run: npx a1 axe http://localhost:3000/checkout --level AA --fail-on serious
      - run: npx a1 sr walk http://localhost:3000/checkout --sr virtual --out walk.md

      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: screen-reader-walk
          path: walk.md
