import { describe, expect } from 'vitest';
import { test } from 'a11ied/vitest';

const team = test.extend({
   srOptions: { url: 'PRACTICE_URLfixed/' },
});

describe('invite dialog', () => {
   team('opens with its name and moves focus to the first field', ({ sr }) => {
      sr.goTo({ role: 'button', name: 'Invite member' })
         .checkpoint('opened')
         .activate()
         .expect.spoken('dialog, Invite member', { since: 'opened' })
         .and.cursorOn({ role: 'textbox', name: 'Email address' });
   });
});
